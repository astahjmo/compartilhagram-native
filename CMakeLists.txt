cmake_minimum_required(VERSION 3.24)
project(Compartilhagram VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Supported platforms are Linux and Windows")
endif()
if(MSVC)
  # The pinned SDK is a release C++ ABI: use Release/RelWithDebInfo, not /MDd.
  set(CMAKE_CONFIGURATION_TYPES "Release;RelWithDebInfo" CACHE STRING "Build configurations" FORCE)
  if(CMAKE_BUILD_TYPE AND NOT CMAKE_BUILD_TYPE MATCHES "^(Release|RelWithDebInfo)$")
    message(FATAL_ERROR "The Windows WebRTC SDK requires Release or RelWithDebInfo")
  endif()
  add_compile_options(/utf-8 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
include(cmake/WebRTC.cmake)
if(WIN32)
  set(PLATFORM_MEDIA_SOURCES src/SystemAudioWindows.cpp src/AudioProcessTree.h)
  set(PLATFORM_MEDIA_LIBRARIES ole32 uuid mmdevapi)
else()
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  set(PLATFORM_MEDIA_SOURCES src/SystemAudio.cpp src/PortalCapture.cpp src/PortalCapture.h)
  set(PLATFORM_MEDIA_LIBRARIES Qt6::DBus PkgConfig::PIPEWIRE PkgConfig::PULSE)
endif()
if(WIN32)
  find_package(libyuv CONFIG REQUIRED)
  set(LIBYUV_LIBRARY yuv)
  set(LIBYUV_INCLUDE_DIR "${libyuv_INCLUDE_DIRS}")
else()
  find_path(LIBYUV_INCLUDE_DIR libyuv/convert.h REQUIRED)
  find_library(LIBYUV_LIBRARY NAMES yuv libyuv REQUIRED)
endif()
add_library(compartilhagram_network src/ServerClient.cpp src/ServerClient.h)
target_link_libraries(compartilhagram_network PUBLIC Qt6::Network)
target_include_directories(compartilhagram_network PUBLIC src)
add_library(compartilhagram_media src/RtcEngine.cpp src/RtcEngine.h src/AudioMix.cpp src/SystemAudio.h ${PLATFORM_MEDIA_SOURCES} src/CodecStatus.cpp src/CodecStatus.h)
target_link_libraries(compartilhagram_media PUBLIC Qt6::Widgets WebRTC ${PLATFORM_MEDIA_LIBRARIES})
target_include_directories(compartilhagram_media PUBLIC src)
target_include_directories(compartilhagram_media PRIVATE ${LIBYUV_INCLUDE_DIR})
target_link_libraries(compartilhagram_media PRIVATE ${LIBYUV_LIBRARY})
target_link_libraries(compartilhagram_media PRIVATE ${CMAKE_DL_LIBS})
qt_add_executable(compartilhagram src/main.cpp src/MainWindow.cpp src/MainWindow.h)
target_link_libraries(compartilhagram PRIVATE compartilhagram_network compartilhagram_media)
if(WIN32)
  set_target_properties(compartilhagram PROPERTIES WIN32_EXECUTABLE TRUE)
else()
  set_target_properties(compartilhagram PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS compartilhagram RUNTIME DESTINATION bin)
if(WIN32)
  install(FILES "$<TARGET_FILE:WebRTC>" DESTINATION bin)
  get_target_property(_qt_qmake Qt6::qmake IMPORTED_LOCATION)
  get_filename_component(_qt_bin "${_qt_qmake}" DIRECTORY)
  find_program(WINDEPLOYQT_EXECUTABLE windeployqt HINTS "${_qt_bin}" REQUIRED)
  configure_file(cmake/DeployWindows.cmake.in DeployWindows.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployWindows.cmake")
else()
  install(FILES "${WEBRTC_ROOT}/lib/libwebrtc.so" DESTINATION lib)
endif()
install(FILES "${WEBRTC_ROOT}/LICENSE" DESTINATION share/compartilhagram RENAME libwebrtc-LICENSE)
if(NOT WIN32)
  install(FILES packaging/compartilhagram.desktop DESTINATION share/applications)
  install(FILES packaging/compartilhagram.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  add_custom_target(appimage
    COMMAND bash "${CMAKE_SOURCE_DIR}/packaging/build-appimage.sh" "${CMAKE_BINARY_DIR}"
    DEPENDS compartilhagram
    USES_TERMINAL
    COMMENT "Bundling Compartilhagram into a single portable AppImage")
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(protocol_test tests/ProtocolTest.cpp)
  target_link_libraries(protocol_test PRIVATE compartilhagram_network Qt6::Test)
  add_test(NAME protocol COMMAND protocol_test)
  add_executable(media_test tests/MediaTest.cpp)
  target_link_libraries(media_test PRIVATE compartilhagram_media Qt6::Test)
  add_test(NAME media COMMAND media_test)
  set_tests_properties(media PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;COMPARTILHAGRAM_DISABLE_GPU=1" TIMEOUT 40)
  add_executable(audio_test tests/AudioTest.cpp)
  target_link_libraries(audio_test PRIVATE compartilhagram_media Qt6::Test)
  if(WIN32)
    add_executable(windows_audio_tone tests/WindowsAudioTone.cpp)
    set_target_properties(windows_audio_tone PROPERTIES AUTOMOC OFF)
    target_link_libraries(windows_audio_tone PRIVATE ole32 uuid)
    add_dependencies(audio_test windows_audio_tone)
    add_test(NAME audio COMMAND audio_test)
  else()
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_test(NAME audio COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:audio_test>)
    add_executable(portal_test tests/PortalTest.cpp)
    target_link_libraries(portal_test PRIVATE compartilhagram_media Qt6::Test)
    add_test(NAME portal COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:portal_test>)
    set_tests_properties(portal PROPERTIES TIMEOUT 20)
  endif()
  set_tests_properties(audio PROPERTIES TIMEOUT 40)
  add_executable(ui_test tests/UiTest.cpp src/MainWindow.cpp src/MainWindow.h)
  target_link_libraries(ui_test PRIVATE compartilhagram_network compartilhagram_media Qt6::Test)
  add_test(NAME ui COMMAND ui_test)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    foreach(test_target protocol_test media_test audio_test ui_test)
      add_custom_command(TARGET ${test_target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "$<TARGET_FILE:WebRTC>" "$<TARGET_FILE_DIR:${test_target}>")
    endforeach()
  endif()
endif()

if(WIN32)
  add_custom_command(TARGET compartilhagram POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:WebRTC>" "$<TARGET_FILE_DIR:compartilhagram>")
endif()
